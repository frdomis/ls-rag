server {
    listen 80;  nginx监听80端口
    server_name _;   不管用户用那个域名来访问。这个配置都接者
    client_max_body_size 10M;   用户上传问价最大10m

    location / {    所有路径都遵守下面规则
        proxy_pass http://127.0.0.1:8000;  请求原样转给8000端口的uvicorn
        proxy_http_version 1.1;   用http协议与后端通信
        proxy_set_header Host $host;   告诉 uvicorn 用户访问的是哪个域名/IP
        proxy_set_header X-Real-IP $remote_addr;   告诉 uvicorn 用户的真实 IP。
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;    记录完整的代理链 IP。
        proxy_read_timeout 120s;   RAG 问答可能比较慢（检索 + 调大模型），把超时放宽到 120 秒
        proxy_buffering off;    关掉缓冲和缓存。
        proxy_cache off;
    }
}
EOF

#PowerShell 调接口输出乱码修改
function Invoke-Api {  函数名
    param([string]$Uri, [hashtable]$Body) $Uri字符串接口地址  $Body哈希表要发的内容
    $json = $Body | ConvertTo-Json -Compress  哈希表转json文本  压成一行
    $bytes = [System.Text.Encoding]::UTF8.GetBytes($json) 把json字符转按utf-8编码成字节数组
    $resp = Invoke-WebRequest -Uri $Uri -Method Post -Body $bytes `
        -ContentType "application/json; charset=utf-8" -UseBasicParsing
    发送post请求   关掉老式html解析引擎的依赖
    $text = [System.Text.Encoding]::UTF8.GetString($resp.RawContentStream.ToArray())
    return $text | ConvertFrom-Json
    反过来再编一编：相应原始字节按utf-8解析成字符串，再把json文本转成powershall对象，
}
